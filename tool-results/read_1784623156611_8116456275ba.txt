   357→function LoginPage() {
   358→  const [email, setEmail] = useState('')
   359→  const [password, setPassword] = useState('')
   360→  const [showPassword, setShowPassword] = useState(false)
   361→  const [isLoading, setIsLoading] = useState(false)
   362→  const [error, setError] = useState('')
   363→  const [selectedDept, setSelectedDept] = useState('CSE')
   364→  const [showDeptDropdown, setShowDeptDropdown] = useState(false)
   365→  const login = useAuthStore((state) => state.login)
   366→
   367→  const handleSubmit = async (e: React.FormEvent) => {
   368→    e.preventDefault()
   369→    setError('')
   370→    setIsLoading(true)
   371→
   372→    try {
   373→      const result = await login(email, password)
   374→      if (!result.success) {
   375→        setError(result.error || 'Invalid email or password')
   376→      }
   377→    } catch (err) {
   378→      setError('Network error. Please try again.')
   379→    } finally {
   380→      setIsLoading(false)
   381→    }
   382→  }
   383→
   384→  const quickLogin = (emailVal: string, passVal: string) => {
   385→    setEmail(emailVal)
   386→    setPassword(passVal)
   387→  }
   388→
   389→  const getDeptEmail = (deptCode: string, role: string) => {
   390→    const code = deptCode.toLowerCase()
