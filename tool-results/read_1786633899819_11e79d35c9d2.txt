   521→function getMonthName(month: number): string {
   522→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 
   523→                  'July', 'August', 'September', 'October', 'November', 'December']
   524→  return months[month - 1] || ''
   525→}

   527→function calculateOverallPerformance(achievements: any[], awards: any[], placements: any[], research: any[]): number {
   528→  const baseScore = (achievements.length * 2) + (awards.length * 3) + (placements.length * 4) + (research.length * 5)
   529→  const maxPossible = 1000
   530→  return Math.min(100, Math.round((baseScore / maxPossible) * 100))
   531→}
   532→
   533→function calculateDeptPerformance(achievements: any[], awards: any[], placements: any[], research: any[], studentCount: number): number {
   534→  if (studentCount === 0) return 0
   535→  const normalizedScore = ((achievements.length * 2) + (awards.length * 3) + (placements.length * 4) + (research.length * 5)) / studentCount
   536→  return Math.min(100, Math.round(normalizedScore * 10))
   537→}

   539→function generateMonthlyTrend(startDate: Date, endDate: Date): any[] {
   540→  const trend = []
   541→  const current = new Date(startDate)
   542→  
   543→  while (current <= endDate) {
   544→    const monthKey = `${current.getFullYear()}-${String(current.getMonth() + 1).padStart(2, '0')}`
   545→    trend.push({
   546→      month: current.toLocaleDateString('en-US', { month: 'short', year: '2-digit' }),
   547→      monthKey,
   548→      year: current.getFullYear(),
   549→      monthNum: current.getMonth() + 1,
   550→      achievements: Math.floor(Math.random() * 50) + 10,
   551→      publications: Math.floor(Math.random() * 20) + 5,
   552→      placements: Math.floor(Math.random() * 30) + 5,
   553→      events: Math.floor(Math.random() * 15) + 3,
   554→      total: 0
   555→    })
   556→    current.setMonth(current.getMonth() + 1)
   557→  }
   558→  
   559→  trend.forEach(t => {
   560→    t.total = t.achievements + t.publications + t.placements + t.events
   561→  })
   562→  
   563→  return trend
   564→}

   566→function generateClassWiseData(students: any[], achievements: any[]): any[] {
   567→  const classMap = new Map()
   568→  
   569→  students.forEach(student => {
   570→    const classKey = `${student.semester || 'Unknown'}-${student.section || 'Default'}`
   571→    if (!classMap.has(classKey)) {
   572→      classMap.set(classKey, {
   573→        semester: student.semester,
   574→        section: student.section,
   575→        studentCount: 0,
   576→        achievementCount: 0,
   577→        achievements: []
   578→      })
   579→    }
   580→    const classData = classMap.get(classKey)
   581→    classData.studentCount++
   582→    classData.achievementCount += student.achievements?.length || 0
   583→    classData.achievements.push(...(student.achievements || []))
   584→  })
   585→  
   586→  return Array.from(classMap.values()).map(c => ({
   587→    ...c,
   588→    avgAchievements: c.studentCount > 0 ? (c.achievementCount / c.studentCount).toFixed(2) : '0',
   589→    performanceScore: c.studentCount > 0 ? Math.min(100, Math.round((c.achievementCount / c.studentCount) * 20)) : 0
   590→  })).sort((a, b) => b.performanceScore - a.performanceScore)
   591→}

   730→function generateResearchByDepartment(research: any[], patents: any[]): any[] {
   731→  const deptMap = new Map()
   732→  research.forEach(r => {
   733→    const dept = r.department?.name || 'Unknown'
   734→    if (!deptMap.has(dept)) {
   735→      deptMap.set(dept, { publications: 0, patents: 0 })
   736→    }
   737→    deptMap.get(dept).publications++
   738→  })
   739→  patents.forEach(p => {
   740→    const dept = p.faculty?.department?.name || 'Unknown'
   741→    if (!deptMap.has(dept)) {
   742→      deptMap.set(dept, { publications: 0, patents: 0 })
   743→    }
   744→    deptMap.get(dept).patents++
   745→  })
   746→  return Array.from(deptMap.entries()).map(([department, data]) => ({ department, ...data }))
   747→}
