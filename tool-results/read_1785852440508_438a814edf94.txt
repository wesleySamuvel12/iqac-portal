   352→// ============ LOGIN PAGE - PREMIUM IMPRESSIVE DESIGN ============
   353→function LoginPage() {
   354→  const [email, setEmail] = useState('')
   355→  const [password, setPassword] = useState('')
   356→  const [showPassword, setShowPassword] = useState(false)
   357→  const [isLoading, setIsLoading] = useState(false)
   358→  const [error, setError] = useState('')
   359→  const [selectedRole, setSelectedRole] = useState<'STUDENT' | 'STAFF' | 'HOD' | 'ADMIN'>('STUDENT')
   360→  const [selectedDept, setSelectedDept] = useState('CSE')
   361→  const [showDeptDropdown, setShowDeptDropdown] = useState(false)
   362→  const [isFocused, setIsFocused] = useState<string | null>(null)
   363→  const [mousePos, setMousePos] = useState({ x: 0, y: 0 })
   364→  const login = useAuthStore((state) => state.login)
   365→
   366→  // Track mouse position for spotlight effect
   367→  useEffect(() => {
   368→    const handleMouseMove = (e: MouseEvent) => {
   369→      setMousePos({ x: e.clientX, y: e.clientY })
   370→    }
   371→    window.addEventListener('mousemove', handleMouseMove)
   372→    return () => window.removeEventListener('mousemove', handleMouseMove)
   373→  }, [])
   374→
   375→  const handleSubmit = async (e: React.FormEvent) => {
   376→    e.preventDefault()
   377→    setError('')
   378→    setIsLoading(true)
   379→
   380→    try {
   381→      const result = await login(email, password)
   382→      if (!result.success) {
   383→        setError(result.error || 'Invalid email or password')
   384→      }
   385→    } catch (err) {
   386→      setError('Network error. Please try again.')
