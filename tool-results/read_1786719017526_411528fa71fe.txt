     6→// Helper function to convert image file to base64 data URL
     7→function imageToBase64DataUrl(filename: string): string {
     8→  const imagePath = join(process.cwd(), 'public/images', filename)
     9→  try {
    10→    if (existsSync(imagePath)) {
    11→      const imageBuffer = readFileSync(imagePath)
    12→      const base64 = imageBuffer.toString('base64')
    13→      // Determine MIME type from extension
    14→      const ext = filename.split('.').pop()?.toLowerCase()
    15→      const mimeType = ext === 'png' ? 'image/png' : ext === 'jpg' || ext === 'jpeg' ? 'image/jpeg' : 'image/png'
    16→      return `data:${mimeType};base64,${base64}`
    17→    }
    18→  } catch (error) {
    19→    console.error(`Error loading logo ${filename}:`, error)
    20→  }
    21→  return ''
    22→}

    24→export async function POST(request: NextRequest) {
    25→  let tempHtmlPath = ''
    26→  let tempPdfPath = ''
    27→  
    28→  try {
    29→    const body = await request.json()
    30→    const reportData = body.reportData || {}
    31→    const department = body.department || ''
    32→
    33→    // Convert logo images to base64 data URLs for embedding in PDF
    34→    const nietLogoDataUrl = imageToBase64DataUrl('niet-logo.png')
    35→    const nehrugroupLogoDataUrl = imageToBase64DataUrl('nehrugroup-logo.png')
    36→
    37→    // Generate HTML content for the PDF with embedded logos
    38→    const htmlContent = generateReportHTML(reportData, department, nietLogoDataUrl, nehrugroupLogoDataUrl)
    39→
    40→    // Create temp directory if not exists
    41→    const tempDir = join(process.cwd(), 'temp')
    42→    if (!existsSync(tempDir)) {
    43→      mkdirSync(tempDir, { recursive: true })
    44→    }
    45→
    46→    // Write HTML to temp file
    47→    const timestamp = Date.now()
    48→    tempHtmlPath = join(tempDir, `report_${timestamp}.html`)
    49→    tempPdfPath = join(tempDir, `report_${timestamp}.pdf`)
    50→    
    51→    writeFileSync(tempHtmlPath, htmlContent)
    52→
    53→    // Try to generate PDF using Playwright (most reliable method)
    54→    try {
    55→      await generateWithPlaywright(htmlContent, tempPdfPath)
    56→    } catch (playwrightError) {
    57→      console.error('Playwright failed:', playwrightError)
    58→      throw new Error('PDF generation failed: ' + (playwrightError as Error).message)
    59→    }
    60→
    61→    // Check if PDF was created
    62→    if (!existsSync(tempPdfPath)) {
    63→      throw new Error('PDF file was not generated')
    64→    }
    65→
    66→    // Read the generated PDF
    67→    const pdfBuffer = readFileSync(tempPdfPath)
    68→
    69→    // Clean up temp files
    70→    try {
    71→      if (existsSync(tempHtmlPath)) unlinkSync(tempHtmlPath)
    72→      if (existsSync(tempPdfPath)) unlinkSync(tempPdfPath)
    73→    } catch (e) {
    74→      console.error('Cleanup error:', e)
    75→    }
    76→
    77→    // Return PDF as downloadable file
    78→    return new NextResponse(pdfBuffer, {
    79→      headers: {
    80→        'Content-Type': 'application/pdf',
    81→        'Content-Disposition': `attachment; filename="Monthly_Department_Report_${department || 'NIET'}_${reportData?.reportingMonth || 'Report'}_${reportData?.reportingYear || new Date().getFullYear()}.pdf"`,
    82→        'Content-Length': pdfBuffer.length.toString(),
    83→      },
    84→    })
    85→  } catch (error) {
    86→    console.error('PDF generation error:', error)
    87→    
    88→    // Cleanup on error
    89→    try {
    90→      if (tempHtmlPath && existsSync(tempHtmlPath)) unlinkSync(tempHtmlPath)
    91→      if (tempPdfPath && existsSync(tempPdfPath)) unlinkSync(tempPdfPath)
    92→    } catch (e) {}
    93→    
    94→    return NextResponse.json({ error: 'Failed to generate PDF: ' + (error as Error).message }, { status: 500 })
    95→  }
    96→}

    98→async function generateWithPlaywright(htmlContent: string, outputPath: string): Promise<void> {
    99→  // Dynamic import for playwright
   100→  const { chromium } = await import('playwright')
   101→  
   102→  const browser = await chromium.launch({
   103→    headless: true,
   104→    args: ['--no-sandbox', '--disable-setuid-sandbox']
   105→  })
   106→  
   107→  try {
   108→    const page = await browser.newPage()
   109→    await page.setContent(htmlContent, { waitUntil: 'networkidle' })
   110→    
   111→    await page.pdf({
   112→      path: outputPath,
   113→      format: 'A4',
   114→      printBackground: true,
   115→      margin: {
   116→        top: '12mm',
   117→        bottom: '12mm',
   118→        left: '12mm',
   119→        right: '12mm'
   120→      },
   121→      displayHeaderFooter: true,
   122→      headerTemplate: '<div></div>',
   123→      footerTemplate: '<div style="font-size: 8px; text-align: center; width: 100%; color: #9ca3af;"><span class="pageNumber"></span> / <span class="totalPages"></span></div>'
   124→    })
   125→  } finally {
   126→    await browser.close()
   127→  }
   128→}
