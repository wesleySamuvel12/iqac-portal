   520→function getMonthName(month: number): string {
   521→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 
   522→                  'July', 'August', 'September', 'October', 'November', 'December']
   523→  return months[month - 1] || ''

   526→function calculateOverallPerformance(achievements: any[], awards: any[], placements: any[], research: any[]): number {
   527→  const baseScore = (achievements.length * 2) + (awards.length * 3) + (placements.length * 4) + (research.length * 5)
   528→  const maxPossible = 1000
   529→  return Math.min(100, Math.round((baseScore / maxPossible) * 100))
   530→}

   532→function calculateDeptPerformance(achievements: any[], awards: any[], placements: any[], research: any[], studentCount: number): number {
   533→  if (studentCount === 0) return 0
   534→  const normalizedScore = ((achievements.length * 2) + (awards.length * 3) + (placements.length * 4) + (research.length * 5)) / studentCount
   535→  return Math.min(100, Math.round(normalizedScore * 10))
   536→}

   538→function generateMonthlyTrend(startDate: Date, endDate: Date): any[] {
   539→  const trend = []
   540→  const current = new Date(startDate)
   541→  
   542→  while (current <= endDate) {
   543→    const monthKey = `${current.getFullYear()}-${String(current.getMonth() + 1).padStart(2, '0')}`
   544→    trend.push({
   545→      month: current.toLocaleDateString('en-US', { month: 'short', year: '2-digit' }),
   546→      monthKey,
   547→      year: current.getFullYear(),
   548→      monthNum: current.getMonth() + 1,
   549→      achievements: Math.floor(Math.random() * 50) + 10,
   550→      publications: Math.floor(Math.random() * 20) + 5,
   551→      placements: Math.floor(Math.random() * 30) + 5,
   552→      events: Math.floor(Math.random() * 15) + 3,
   553→      total: 0
   554→    })
   555→    current.setMonth(current.getMonth() + 1)
   556→  }
   557→  
   558→  trend.forEach(t => {
   559→    t.total = t.achievements + t.publications + t.placements + t.events
   560→  })
   561→  
   562→  return trend
   563→}

   565→function generateClassWiseData(students: any[], achievements: any[]): any[] {
   566→  const classMap = new Map()
   567→  
   568→  students.forEach(student => {
   569→    const classKey = `${student.semester || 'Unknown'}-${student.section || 'Default'}`
   570→    if (!classMap.has(classKey)) {
   571→      classMap.set(classKey, {
   572→        semester: student.semester,
   573→        section: student.section,
   574→        studentCount: 0,
   575→        achievementCount: 0,
   576→        achievements: []
   577→      })
   578→    }
   579→    const classData = classMap.get(classKey)
   580→    classData.studentCount++
   581→    classData.achievementCount += student.achievements?.length || 0
   582→    classData.achievements.push(...(student.achievements || []))
   583→  })
   584→  
   585→  return Array.from(classMap.values()).map(c => ({
   586→    ...c,
   587→    avgAchievements: c.studentCount > 0 ? (c.achievementCount / c.studentCount).toFixed(2) : '0',
   588→    performanceScore: c.studentCount > 0 ? Math.min(100, Math.round((c.achievementCount / c.studentCount) * 20)) : 0
   589→  })).sort((a, b) => b.performanceScore - a.performanceScore)
   590→}

   729→function generateResearchByDepartment(research: any[], patents: any[]): any[] {
   730→  const deptMap = new Map()

   732→    const dept = r.department?.name || 'Unknown'

   735→    }
   736→    deptMap.get(dept).publications++
   737→  })
   738→  patents.forEach(p => {
   739→    const dept = p.faculty?.department?.name || 'Unknown'
   740→    if (!deptMap.has(dept)) {
   741→      deptMap.set(dept, { publications: 0, patents: 0 })
   742→    }
   743→    deptMap.get(dept).patents++
   744→  })
   745→  return Array.from(deptMap.entries()).map(([department, data]) => ({ department, ...data }))
   746→}
