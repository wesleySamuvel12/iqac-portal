   354→// ============ LOGIN PAGE ============
   355→function LoginPage() {
   356→  const [email, setEmail] = useState('')
   357→  const [password, setPassword] = useState('')
   358→  const [showPassword, setShowPassword] = useState(false)
   359→  const [isLoading, setIsLoading] = useState(false)
   360→  const [error, setError] = useState('')
   361→  const [selectedDept, setSelectedDept] = useState('CSE')
   362→  const [showDeptDropdown, setShowDeptDropdown] = useState(false)
   363→  const login = useAuthStore((state) => state.login)
   364→
   365→  const handleSubmit = async (e: React.FormEvent) => {
   366→    e.preventDefault()
   367→    setError('')
   368→    setIsLoading(true)
   369→
   370→    try {
   371→      const result = await login(email, password)
   372→      if (!result.success) {
   373→        setError(result.error || 'Invalid email or password')
   374→      }
   375→    } catch (err) {
   376→      setError('Network error. Please try again.')
   377→    } finally {
   378→      setIsLoading(false)
   379→    }
   380→  }
   381→
   382→  const quickLogin = (emailVal: string, passVal: string) => {
   383→    setEmail(emailVal)
   384→    setPassword(passVal)
   385→  }
   386→
   387→  const getDeptEmail = (deptCode: string, role: string) => {
   388→    const code = deptCode.toLowerCase()
   389→    switch(role) {
   390→      case 'HOD': return `hod_${code}@niet.ac.in`
