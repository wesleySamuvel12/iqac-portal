   492→function getMonthName(month: number): string {
   493→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 
   494→                  'July', 'August', 'September', 'October', 'November', 'December']
   495→  return months[month - 1] || ''

   498→function calculateOverallPerformance(achievements: any[], awards: any[], placements: any[], research: any[]): number {
   499→  const baseScore = (achievements.length * 2) + (awards.length * 3) + (placements.length * 4) + (research.length * 5)
   500→  const maxPossible = 1000
   501→  return Math.min(100, Math.round((baseScore / maxPossible) * 100))
   502→}

   504→function calculateDeptPerformance(achievements: any[], awards: any[], placements: any[], research: any[], studentCount: number): number {
   505→  if (studentCount === 0) return 0
   506→  const normalizedScore = ((achievements.length * 2) + (awards.length * 3) + (placements.length * 4) + (research.length * 5)) / studentCount
   507→  return Math.min(100, Math.round(normalizedScore * 10))
   508→}

   510→function generateMonthlyTrend(startDate: Date, endDate: Date): any[] {
   511→  const trend = []
   512→  const current = new Date(startDate)
   513→  
   514→  while (current <= endDate) {
   515→    const monthKey = `${current.getFullYear()}-${String(current.getMonth() + 1).padStart(2, '0')}`
   516→    trend.push({
   517→      month: current.toLocaleDateString('en-US', { month: 'short', year: '2-digit' }),
   518→      monthKey,
   519→      year: current.getFullYear(),
   520→      monthNum: current.getMonth() + 1,
   521→      achievements: Math.floor(Math.random() * 50) + 10,
   522→      publications: Math.floor(Math.random() * 20) + 5,
   523→      placements: Math.floor(Math.random() * 30) + 5,
   524→      events: Math.floor(Math.random() * 15) + 3,
   525→      total: 0
   526→    })
   527→    current.setMonth(current.getMonth() + 1)
   528→  }
   529→  
   530→  trend.forEach(t => {
   531→    t.total = t.achievements + t.publications + t.placements + t.events
   532→  })
   533→  
   534→  return trend
   535→}

   537→function generateClassWiseData(students: any[], achievements: any[]): any[] {
   538→  const classMap = new Map()
   539→  
   540→  students.forEach(student => {
   541→    const classKey = `${student.semester || 'Unknown'}-${student.section || 'Default'}`
   542→    if (!classMap.has(classKey)) {
   543→      classMap.set(classKey, {
   544→        semester: student.semester,
   545→        section: student.section,
   546→        studentCount: 0,
   547→        achievementCount: 0,
   548→        achievements: []
   549→      })
   550→    }
   551→    const classData = classMap.get(classKey)
   552→    classData.studentCount++
   553→    classData.achievementCount += student.achievements?.length || 0
   554→    classData.achievements.push(...(student.achievements || []))
   555→  })
   556→  
   557→  return Array.from(classMap.values()).map(c => ({
   558→    ...c,
   559→    avgAchievements: c.studentCount > 0 ? (c.achievementCount / c.studentCount).toFixed(2) : '0',
   560→    performanceScore: c.studentCount > 0 ? Math.min(100, Math.round((c.achievementCount / c.studentCount) * 20)) : 0
   561→  })).sort((a, b) => b.performanceScore - a.performanceScore)
   562→}

   564→function generateTopStudentsData(students: any[], achievements: any[]): any[] {
   565→  return students
   566→    .map(student => ({
   567→      id: student.id,
   568→      registerNumber: student.registerNumber,
   569→      name: student.user?.name || student.name,
   570→      department: student.department?.name,
   571→      semester: student.semester,
   572→      section: student.section,
   573→      cgpa: student.cgpa || 0,
   574→      achievementCount: student.achievements?.length || 0,
   575→      achievements: student.achievements?.slice(0, 10) || [],
   576→      score: calculateStudentScore(student)
   577→    }))
   578→    .sort((a, b) => b.score - a.score)
   579→    .slice(0, 20)
   580→}

   582→function calculateStudentScore(student: any): number {
   583→  const cgpaScore = (student.cgpa || 0) * 10
   584→  const achievementScore = (student.achievements?.length || 0) * 5
   585→  const placementScore = student._count?.placements ? student._count.placements * 20 : 0
   586→  return cgpaScore + achievementScore + placementScore
   587→}

   701→function generateResearchByDepartment(research: any[], patents: any[]): any[] {
   702→  const deptMap = new Map()
   703→  research.forEach(r => {
   704→    const dept = r.department?.name || 'Unknown'
   705→    if (!deptMap.has(dept)) {
   706→      deptMap.set(dept, { publications: 0, patents: 0 })
   707→    }
   708→    deptMap.get(dept).publications++
   709→  })
   710→  patents.forEach(p => {
   711→    const dept = p.faculty?.department?.name || 'Unknown'
   712→    if (!deptMap.has(dept)) {
   713→      deptMap.set(dept, { publications: 0, patents: 0 })
   714→    }
   715→    deptMap.get(dept).patents++
   716→  })
   717→  return Array.from(deptMap.entries()).map(([department, data]) => ({ department, ...data }))
   718→}
