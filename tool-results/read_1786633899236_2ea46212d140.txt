    85→// Helper function to create bordered cell
    86→function createCell(
    87→  text: string,
    88→  options: {
    89→    width?: number
    90→    bold?: boolean
    91→    shading?: string
    92→    alignment?: typeof AlignmentType[keyof typeof AlignmentType]
    93→    verticalAlign?: typeof VerticalAlign[keyof typeof VerticalAlign]
    94→    rowSpan?: number
    95→    colSpan?: number
    96→    fontSize?: number
    97→  } = {}
    98→): TableCell {
    99→  const {
   100→    width = 1000,
   101→    bold = false,
   102→    shading,
   103→    alignment = AlignmentType.LEFT,
   104→    verticalAlign = VerticalAlign.CENTER,
   105→    rowSpan = 1,
   106→    colSpan = 1,
   107→    fontSize = 20
   108→  } = options
   109→
   110→  return new TableCell({
   111→    width: { size: width, type: WidthType.DXA },
   112→    rowSpan,
   113→    columnSpan: colSpan,
   114→    shading: shading ? { type: ShadingType.SOLID, color: shading, fill: shading } : undefined,
   115→    verticalAlign,
   116→    borders: {
   117→      top: { style: BorderStyle.SINGLE, size: 1, color: '000000' },
   118→      bottom: { style: BorderStyle.SINGLE, size: 1, color: '000000' },
   119→      left: { style: BorderStyle.SINGLE, size: 1, color: '000000' },
   120→      right: { style: BorderStyle.SINGLE, size: 1, color: '000000' },
   121→    },
   122→    children: [
   123→      new Paragraph({
   124→        alignment,
   125→        children: [
   126→          new TextRun({
   127→            text: text || '',
   128→            bold,
   129→            size: fontSize,
   130→            font: 'Times New Roman',
   131→          }),
   132→        ],
   133→      }),
   134→    ],
   135→  })
   136→}
   137→
   138→// Helper function to create header cell with background color
   139→function createHeaderCell(
   140→  text: string,
   141→  options: {
   142→    width?: number
   143→    colSpan?: number
   144→    rowSpan?: number
   145→    color?: string
   146→    fontSize?: number
   147→  } = {}
   148→): TableCell {
   149→  return createCell(text, {
   150→    ...options,
   151→    bold: true,
   152→    shading: options.color || 'D4EDDA',
   153→    alignment: AlignmentType.CENTER,
   154→    fontSize: options.fontSize || 18,
   155→  })
   156→}
