   358→function LoginPage() {
   359→  const [email, setEmail] = useState('')
   360→  const [password, setPassword] = useState('')
   361→  const [showPassword, setShowPassword] = useState(false)
   362→  const [isLoading, setIsLoading] = useState(false)
   363→  const [error, setError] = useState('')
   364→  const [selectedDept, setSelectedDept] = useState('CSE')
   365→  const [showDeptDropdown, setShowDeptDropdown] = useState(false)
   366→  const login = useAuthStore((state) => state.login)
   367→
   368→  const handleSubmit = async (e: React.FormEvent) => {
   369→    e.preventDefault()
   370→    setError('')
   371→    setIsLoading(true)
   372→
   373→    try {
   374→      const result = await login(email, password)
   375→      if (!result.success) {
   376→        setError(result.error || 'Invalid email or password')
   377→      }
   378→    } catch (err) {
   379→      setError('Network error. Please try again.')
   380→    } finally {
   381→      setIsLoading(false)
   382→    }
   383→  }
   384→
   385→  const quickLogin = (emailVal: string, passVal: string) => {
   386→    setEmail(emailVal)
   387→    setPassword(passVal)
   388→  }
   389→
   390→  const getDeptEmail = (deptCode: string, role: string) => {
   391→    const code = deptCode.toLowerCase()
